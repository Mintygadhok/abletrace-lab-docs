===== Trace_MaterialDetails_SP (PROCEDURE) =====
BEGIN
	SET @receiveLotID := receiveLotID;
	SET @materialID := materialID;
	SET @companyID := companyID;
    
	CREATE TEMPORARY TABLE temp_table (
		mlc_id VARCHAR(100),
		mo_number VARCHAR(100),
		product_lot_code VARCHAR(100),
		qty VARCHAR(100),
		received_qty VARCHAR(100),
		formula_internal_code varchar(100),
		formula_title varchar(100),
		unit_name varchar(50),
		wgt_kgs_per_unit varchar(50)
	);

	create temporary table temp_qty_allocated (
		mlc_id varchar(100),
		qty_allocated varchar(100),
		material_id varchar(100),
		uom varchar(100),
		unit_name varchar(100)
	);

	create temporary table temp_qty_returned (
		mlc_id varchar(100),
		qty_returned varchar(100)
	);	

	SET @returnedQty := (SELECT SUM(qty_return) as qty_return from returnmpreceivelots WHERE material_id = @materialID);

	SET @formulationIDs := (SELECT GROUP_CONCAT(`fosubrecipe__sub_recipe_id`.`formulation_id`) 
	from `subrecipematerials` left outer join `fosubrecipe` as `fosubrecipe__sub_recipe_id` on `subrecipematerials`.`sub_recipe_id` = `fosubrecipe__sub_recipe_id`.`id` 
	where `subrecipematerials`.`material_id` = @materialID);  -- this will give a sub_receipe_id for given material_id
    
    SET @formulationPackagingID := (SELECT GROUP_CONCAT(formulation_id) 
	from `fopackaging` 
	where `fopackaging`.`material_id` = @materialID);  -- this will give a formulation_id for given material_id
    
    SET @allFormulationIDs := TRIM(BOTH ',' FROM CONCAT_WS(',', @formulationIDs, @formulationPackagingID));

	INSERT into temp_table(mlc_id, mo_number, product_lot_code, qty, received_qty, formula_internal_code, formula_title, unit_name, wgt_kgs_per_unit)
	select mlomanagement.id, mlomanagement.internalCode, lotCode, mlomanagement.qty, mlomanagement.received_qty, formulations__formula_id.internalCode, formulations__formula_id.title, unitmeasurement.unit_name, (SELECT fopackaging.wgt_kgs_per_unit FROM mlcpackaging left join fopackaging on mlcpackaging.pack_level_id = fopackaging.id WHERE mlcpackaging.mlc_id = mlomanagement.id) as wgt_kgs_per_unit
	from `mlomanagement` 
	left outer join `mlo_status_types` as `mlo_status_types__mlc_status` on `mlomanagement`.`mlc_status` = `mlo_status_types__mlc_status`.`id`
	left outer join `formulations` as `formulations__formula_id` on `mlomanagement`.`formula_id` = `formulations__formula_id`.`id` 
	left outer join `unitmeasurement` on formulations__formula_id.uom = `unitmeasurement`.`id`
	where `mlomanagement`.`company_id` = @companyID and FIND_IN_SET(`mlomanagement`.`formula_id`, @allFormulationIDs);

	SET @mlcIDs := (SELECT GROUP_CONCAT(mlc_id) FROM temp_table);
	-- SELECT @mlcIDs;

	SET @mprIDs := (SELECT GROUP_CONCAT(id) FROM materialsproductsreleased WHERE FIND_IN_SET(mlc_id, @mlcIDs));
	INSERT into temp_qty_allocated(mlc_id, qty_allocated, material_id, uom, unit_name)
	SELECT materialsproductsreleased.mlc_id, mprrecievelots.qty_allocated, mprrecievelots.material_id, materials.uom, unitmeasurement.unit_name
	FROM mprrecievelots 
	left join materialsproductsreleased on mprrecievelots.MPR_id = materialsproductsreleased.id
	left join materials on mprrecievelots.material_id = materials.id
	left join unitmeasurement on materials.uom = unitmeasurement.id
	WHERE Rec_Lot_id = @receiveLotID;

	-- This is the code to get returned material
	SET @rmpIDs := (SELECT GROUP_CONCAT(id) FROM returnmaterialproduct WHERE FIND_IN_SET(mlc_id, @mlcIDs));
	INSERT into temp_qty_returned(mlc_id, qty_returned)
	SELECT returnmaterialproduct.mlc_id, returnmpreceivelots.qty_return
	FROM returnmpreceivelots 
	left join returnmaterialproduct on returnmpreceivelots.ReturnMP_id = returnmaterialproduct.id 
	WHERE FIND_IN_SET(`returnmpreceivelots`.`ReturnMP_id`, @rmpIDs) AND Rec_Lot_id = @receiveLotID;

	-- SELECT * FROM temp_table;
	SELECT temp_table.*, COALESCE(temp_qty_allocated.qty_allocated, 0) as qty_allocated, temp_qty_allocated.unit_name as qty_allocated_unit, (SELECT COALESCE(SUM(qty_returned), 0) from temp_qty_returned WHERE mlc_id = temp_table.mlc_id) as qty_returned FROM temp_qty_allocated 
	left join temp_table on temp_qty_allocated.mlc_id = temp_table.mlc_id;
	DROP TEMPORARY TABLE temp_table;
	DROP TEMPORARY TABLE temp_qty_allocated;
	DROP TEMPORARY TABLE temp_qty_returned;
END


===== Trace_ProductOneStepBackwardIP_SP (PROCEDURE) =====
BEGIN
	SET @formulationId := formulationId;
	SET @mlcId := mlcId;
    
	select * from 
(
SELECT 
		mpr.mlc_id as mlc_id,
        f.title AS int_prod,
        f.internalCode AS prod_id_int,
        f.myCode AS prod_id_ext,
        mlo.internalCode AS mo_no,
        mlo.lotCode AS prod_lot_code,
        mpr1.qty_allocated AS qty_used,
        mpr1.qty_allocated / fo.wgt_kgs_per_unit AS shipping_units,
        um.unit_name
    FROM 
        materialsproductsreleased mpr
    JOIN 
        mprrecievelots mpr1 ON mpr.id = mpr1.MPR_id
    JOIN 
        receiveproducts rp ON mpr1.Rec_Product_id = rp.id
    JOIN 
        mlomanagement mlo ON mlo.id = rp.mlc_id
    JOIN 
        formulations f ON f.id = mpr1.formula_id
    JOIN 
        fopackaging fo ON fo.formulation_id = f.id
    JOIN 
        unitmeasurement um ON um.id = f.uom
) t
where mlc_id = @mlcId;
END

BEGIN

	SET @mlcId := mlcId;

SELECT f.title as int_prod
, f.internalCode as prod_id_int
, f.myCode as prod_id_ext
, mlo.internalCode

JOIN receiveproducts rp ON mpr.Rec_Product_id = rp.id

===== Trace_ProductOneStepForward_SP (PROCEDURE) =====
BEGIN
    DECLARE v_formulationId VARCHAR(100);
    DECLARE v_mlcId VARCHAR(100);

    SET v_formulationId = formulationId;
    SET v_mlcId = mlcId;

    SELECT 
        c.customer_name,
        s.customer_shipping_address,
        c.address,
        ps.internalCode,
        DATE(ps.createdAt) AS ps_create_date,
        DATE(ps.shippingdate) AS ps_shipping_date,
        s.internalCode AS so_number,
        ps.vehicle_no AS customer_ship_ref,
        psd.shipped_qty * fop.wgt_kgs_per_unit AS shipped_qty_weight,
        psd.shipped_qty AS shipped_qty_units,
        u.unit_name
    FROM 
        dispatchorders do
    JOIN fopackaging fop ON do.packing_id = fop.id
    JOIN formulations f ON do.formula_id = f.id
    JOIN unitmeasurement u ON f.uom = u.id
    JOIN do_receive_products drp ON drp.dispatch_order_id = do.id
    JOIN receiveproducts rp ON drp.recProduct_id = rp.id
    JOIN somanagement s ON do.SO_id = s.id
    JOIN companycustomers c ON s.customer_id = c.id AND s.company_id = c.company_id
    JOIN packingslipdos psd ON do.id = psd.DO_id
    JOIN packingslips ps ON ps.id = psd.PS_id
    WHERE 
        do.formula_id = v_formulationId
        AND rp.mlc_id = v_mlcId
        AND ps.shippingdate IS NOT NULL;
END

	left outer join `returnmaterialproduct` on `mlomanagement`.`id` = `returnmaterialproduct`.`mlc_id`
	left outer join `user` on `mlomanagement`.`startedBy_userId` = `user`.`id`
	where `mlomanagement`.`id` = mlcId and `mlomanagement`.`status_id` = statusId;
END


===== WhC_GetMoPackagingConfiguration_SP (PROCEDURE) =====
BEGIN
	SELECT `fopackaging`.`id`, `fopackaging`.`wgt_kgs_per_unit`, `fopackaging`.`material_id`, `fopackaging`.`whd_flag`, `fopackaging`.`pack_level`, `fopackaging`.`quantity`,
	`materials`.`title` as `materials_title`, `materials`.`uom` as `materials_uom`, `materials`.`statutory_flag` as `materials_statutory_flag`, `materials`.`statutory_file_name` as `materials_statutory_file_name`,
	`unitmeasurement_materials`.`unit_name` as `materials_unit_name`,
	`unitmeasurement_formulations`.`unit_name` as `formulations_unit_name`
	FROM `fopackaging`
	left outer join `materials` on `fopackaging`.`material_id` = `materials`.`id` 
	left outer join `unitmeasurement` as `unitmeasurement_materials` on `materials`.`uom` = `unitmeasurement_materials`.`id`
	left outer join `formulations` on `fopackaging`.`formulation_id` = `formulations`.`id` 
	left outer join `unitmeasurement` as `unitmeasurement_formulations` on `formulations`.`uom` = `unitmeasurement_formulations`.`id`
	where `fopackaging`.`formulation_id` = formulationId;
END

END
